// Global tracking variables for the device connection layer let engoRxCharacteristic = null; let strydServiceChar = null; // Core Service and Data Channel Blueprint Constants const ENGO_SERVICE_UUID = '0783b03e-8535-b5a0-7140-a304d2495cb7'; const ENGO_CHAR_UUID = '0783b03e-8535-b5a0-7140-a304d24a5cb7'; const RSC_SERVICE_UUID = 'running_speed_and_cadence'; // Standard Bluetooth SIG Profile const RSC_CHAR_UUID = 'rsc_measurement'; /** * Visual feedback logger to keep you updated on progress directly in the web page interface. */ function updateStatus(msg) { const statusElement = document.getElementById('status'); if (statusElement) { statusElement.innerText = msg; } console.log("[Bridge Log]: " + msg); } /** * Initializes the dual Bluetooth connection pipeline. * Binds your phone as a Central BLE hub talking to both endpoints simultaneously. */ async function initializeBridge() { try { // 1. Scan and link to your Engo 3 Smart Glasses updateStatus("Scanning for Engo 3 Glasses..."); const engoDevice = await navigator.bluetooth.requestDevice({ filters: [ { namePrefix: 'ENGO' }, { namePrefix: 'engo' } ], optionalServices: [ENGO_SERVICE_UUID] }); updateStatus("Glasses found. Connecting to ActiveLook engine..."); const engoServer = await engoDevice.gatt.connect(); const engoService = await engoServer.getPrimaryService(ENGO_SERVICE_UUID); engoRxCharacteristic = await engoService.getCharacteristic(ENGO_CHAR_UUID); // 2. Scan and link to your Stryd Footpod with broad fallback validation profiles updateStatus("Engo connected! Scanning for Stryd Footpod (Shake pod to wake)..."); const strydDevice = await navigator.bluetooth.requestDevice({ filters: [ { namePrefix: 'Stryd' }, { namePrefix: 'STRYD' }, { namePrefix: 'stryd' } ], optionalServices: [ RSC_SERVICE_UUID, '00001814-0000-1000-8000-00805f9b34fb', // Explicit 16-bit RSC Service layout 'cycling_power' // Alternative metric broadcast layout ] }); updateStatus("Footpod found. Accessing Running Speed & Cadence characteristics..."); const strydServer = await strydDevice.gatt.connect(); // Try getting the service via standard alias name, fallback to numeric short-string if needed let strydService; try { strydService = await strydServer.getPrimaryService(RSC_SERVICE_UUID); } catch (e) { strydService = await strydServer.getPrimaryService('00001814-0000-1000-8000-00805f9b34fb'); } strydServiceChar = await strydService.getCharacteristic(RSC_CHAR_UUID); // 3. Activate the asynchronous sub-second real-time notification stream updateStatus("Binding live telemetry listeners..."); await strydServiceChar.startNotifications(); strydServiceChar.addEventListener('characteristicvaluechanged', processIncomingStrydData); updateStatus("Bridge fully active! Ready to run."); } catch (error) { updateStatus("Connection Failure: " + error.message); updateStatus("Make sure the pod isn't connected to your watch or official app."); } } /** * Handles incoming binary packets coming directly from the Stryd pod hardware layer. * Deciphers the custom flags, calculates true running pace, shifts byte positions, * extracts total miles, grabs the watt value, and updates the canvas. */ function processIncomingStrydData(event) { let dataView = event.target.value; // This is a standard JavaScript DataView object // Byte 0: Feature configuration array flags let flags = dataView.getUint8(0); let strideLengthPresent = flags & 0x01; let totalDistancePresent = flags & 0x02; let byteOffset = 1; // 1. Calculate Live Running Pace // Speed is stored as an unsigned 16-bit integer (2 bytes) scaled by 1/256 in meters per second let rawSpeed = dataView.getUint16(byteOffset, true); byteOffset += 2; let speedMps = rawSpeed / 256; let paceStr = "--:--"; if (speedMps > 0.447) { // Only calculate true pace if moving faster than 1 mph to avoid infinity bounds let mph = speedMps * 2.23694; let decimalPace = 60 / mph; let paceMin = Math.floor(decimalPace); let paceSec = Math.floor((decimalPace - paceMin) * 60).toString().padStart(2, '0'); paceStr = `${paceMin}:${paceSec}`; } // 2. Shift past basic Cadence tracking metrics (1 byte) byteOffset += 1; // 3. Shift past Stride Length if flag index is checked (2 bytes) if (strideLengthPresent) { byteOffset += 2; } // 4. Extract Total Accumulated Run Distance // Distance tracking is stored as a 32-bit unsigned integer (4 bytes) scaled by 1/10 in meters let distanceMiles = "0.00"; if (totalDistancePresent) { let totalMeters = dataView.getUint32(byteOffset, true) / 10; distanceMiles = (totalMeters * 0.000621371).toFixed(2); byteOffset += 4; } // 5. Parse Real-Time Running Power // Stryd safely injects raw power (watts) as a Uint16 at the trailing offset of the standard array payload let powerWatts = 0; if (byteOffset < dataView.byteLength) { powerWatts = dataView.getUint16(byteOffset, true); } // 6. Push values directly onto the phone's web dashboard layout fields document.getElementById('view-pwr').innerText = `${powerWatts} W`; document.getElementById('view-pce').innerText = paceStr; document.getElementById('view-dst').innerText = `${distanceMiles} mi`; // 7. Hand parameters down to the ActiveLook glass rendering encoder pushLayoutToGlasses(powerWatts, paceStr, distanceMiles); } /** * Formats data strings into ActiveLook API clear-and-text parameters * and pushes them out via an active Bluetooth transmission loop. */ async function pushLayoutToGlasses(pwr, pce, dst) { // Gracefully exit if the glasses characteristic hasn't initialized yet if (!engoRxCharacteristic) return; // Assemble layout rows. Text positioning uses a grid structure: text [X-coordinate] [Y-coordinate] "[String]" let commands = `clear\n` + `text 10 20 "PWR: ${pwr} W"\n` + `text 10 50 "PCE: ${pce}"\n` + `text 10 80 "DST: ${dst} mi"\n`; try { let stringEncoder = new TextEncoder(); let encodedBytes = stringEncoder.encode(commands); // Push raw compiled array straight to the glass display plane buffer await engoRxCharacteristic.writeValueWithResponse(encodedBytes); } catch (bleWriteError) { console.warn("Frame transmission dropped: " + bleWriteError.message); } }
