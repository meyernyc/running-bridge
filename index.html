import datetime import pandas as pd import requests from sqlalchemy import func from ..api.database import engine from ..app import app from ..utils import config from .sqlalchemy_declarative import strydSummary def auth_stryd_session(): """Authenticate with Stryd and return (session_token, user_id).""" request_json = { "email": config.get("stryd", "username"), "password": config.get("stryd", "password"), } response_data = requests.post( "https://www.stryd.com/b/email/signin", json=request_json ) if response_data.status_code != 200: app.server.logger.debug("Stryd could not authenticate") raise Exception( f"Failed password authentication. Status: {response_data.status_code}" ) app.server.logger.debug("Stryd authenticated") temp_data = response_data.json() return temp_data["token"], temp_data["id"] ############################## ## Get the list of workouts ## ############################## def pull_stryd_data(): """Pull workout history from Stryd API and store new entries in DB.""" session_id, user_id = auth_stryd_session() # Pass tomorrow's date to avoid timezone boundary truncation today = datetime.datetime.now() + datetime.timedelta(days=1) start = today - datetime.timedelta(days=9999) headers = {"Authorization": f"Bearer {session_id}"} url = "https://www.stryd.com/b/api/v1/activities/calendar" json_data = { "srtDate": start.strftime("%m-%d-%Y"), "endDate": today.strftime("%m-%d-%Y"), "sortBy": "StartDate", } response_data = requests.get(url, headers=headers, params=json_data) if response_data.status_code != 200: raise Exception( f"Failed to fetch calendar data. Status: {response_data.status_code}" ) df = pd.DataFrame( response_data.json().get("activities", []) ) # Summary data for workouts if df.empty: app.server.logger.info("No activities returned from Stryd API.") return df df.rename( columns={"timestamp": "start_date_local", "ftp": "stryd_ftp", "stress": "rss"}, inplace=True, ) df["start_date_local"] = df["start_date_local"].apply( datetime.datetime.fromtimestamp ) df.set_index(pd.to_datetime(df["start_date_local"]), inplace=True) # Filter specific columns to persist columns_to_keep = [ "stryd_ftp", "total_elevation_gain", "total_elevation_loss", "max_elevation", "min_elevation", "average_cadence", "max_cadence", "min_cadence", "average_stride_length", "max_stride_length", "min_stride_length", "average_ground_time", "max_ground_time", "min_ground_time", "average_oscillation", "max_oscillation", "min_oscillation", "average_leg_spring", "rss", "max_vertical_stiffness", "stryds", "elevation", "temperature", "humidity", "windBearing", "windSpeed", "windGust", "dewPoint", ] # Keep only available target columns from API response existing_cols = [col for col in columns_to_keep if col in df.columns] df = df[existing_cols] try: # Filter df for only new records not yet in DB last_stryd_date = app.session.query( func.max(strydSummary.start_date_local) ).scalar() if last_stryd_date: df = df[df.index > last_stryd_date] if len(df) > 0: app.server.logger.info("New Stryd workouts found!") df.to_sql("stryd_summary", engine, if_exists="append", index=True) app.session.commit() finally: app.session.remove() return df def get_training_distribution(race=1, gender=1, age=1): """Fetch training distribution metrics compared to benchmark demographics.""" session_id, user_id = auth_stryd_session() headers = {"Authorization": f"Bearer {session_id}"} url = "https://www.stryd.com/b/api/v1/users/runner-attribute" params = { "race": config.get("stryd", "compare_against_race_event"), "gender": config.get("stryd", "compare_against_gender"), "age": config.get("stryd", "compare_against_age"), } response_data = requests.get(url, headers=headers, params=params) if response_data.status_code != 200: raise Exception( f"Failed to fetch training distribution. Status: {response_data.status_code}" ) return response_data.json()
